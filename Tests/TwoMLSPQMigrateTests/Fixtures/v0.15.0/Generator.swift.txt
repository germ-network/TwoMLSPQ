// MARK: - Recording persistence sink

/// Records the latest bytes pushed to each slot, by seq (persists can arrive out of order
/// per the `PersistenceSink` contract).
final class RecordingSink: PersistenceSink, @unchecked Sendable {
	private let lock = NSLock()
	private var core: (seq: UInt64, bytes: Data)?
	private var checkpoint: (seq: UInt64, bytes: Data)?

	func persist(seq: UInt64, slot: PersistedSlot, bytes: Data) {
		lock.lock()
		defer { lock.unlock() }
		switch slot {
		case .core:
			if core == nil || seq > core!.seq { core = (seq, bytes) }
		case .checkpoint:
			if checkpoint == nil || seq > checkpoint!.seq { checkpoint = (seq, bytes) }
		}
	}

	/// A `PQSession.Persisted` snapshot of whatever has been pushed so far, plus the seq of
	/// each slot (nil core seq if no core has been pushed yet).
	func snapshot() throws -> (persisted: PQSession.Persisted, coreSeq: UInt64?, checkpointSeq: UInt64) {
		lock.lock()
		defer { lock.unlock() }
		guard let checkpoint else {
			throw TestErrors.unexpected  // installSink always pushes a baseline checkpoint
		}
		return (
			PQSession.Persisted(core: core?.bytes, checkpoint: checkpoint.bytes),
			core?.seq, checkpoint.seq
		)
	}
}

// MARK: - Opaque test app-welcome codec

/// A throwaway length-prefixed pack/unpack of the four pieces `PQInvitation.receive` needs,
/// standing in for the app's own identity-envelope format (which lives outside this package).
/// Only this test constructs and parses it — the crate treats the bytes as fully opaque.
enum TestAppWelcomeCodec {
	static func encode(
		sendGroupWelcome: Data, remoteKeyPackage: Data, bootstrapKpCommitment: Data,
		remoteClientId: Data
	) -> Data {
		var out = Data()
		for part in [sendGroupWelcome, remoteKeyPackage, bootstrapKpCommitment, remoteClientId] {
			let len = UInt32(part.count).littleEndian
			withUnsafeBytes(of: len) { out.append(contentsOf: $0) }
			out.append(part)
		}
		return out
	}

	static func decode(_ data: Data) throws -> (
		sendGroupWelcome: Data, remoteKeyPackage: Data, bootstrapKpCommitment: Data,
		remoteClientId: Data
	) {
		var rest = data[...]
		func takePart() throws -> Data {
			guard rest.count >= 4 else { throw TestErrors.unexpected }
			let len = Int(
				rest.prefix(4).withUnsafeBytes { $0.loadUnaligned(as: UInt32.self) }.littleEndian)
			rest = rest.dropFirst(4)
			guard rest.count >= len else { throw TestErrors.unexpected }
			let part = Data(rest.prefix(len))
			rest = rest.dropFirst(len)
			return part
		}
		return (try takePart(), try takePart(), try takePart(), try takePart())
	}
}

// MARK: - meta.json shape

private struct FixtureMeta: Codable {
	struct SideInfo: Codable {
		var coreSeq: UInt64?
		var checkpointSeq: UInt64
		var isFullyEstablished: Bool
		/// `"weInitiate"` / `"theyInitiate"` — this side's own `PQSession.turn`.
		var turn: String
		/// Which PQ side-band round (if any) this side's OWN state reflects as in flight —
		/// `"A.3"`, `"A.4"`, or nil. A side that never received/initiated anything for a round
		/// records nil even if the OTHER side has one parked (the whole point of p4/p5: the
		/// parking is one-sided — the peer is oblivious).
		var roundInFlight: String?
		/// `pendingSideBand(sealing:)` is non-nil for this side at capture — an unconsumed
		/// outbound side-band leg this side still owes delivery of.
		var legParked: Bool
	}
	struct PointInfo: Codable {
		var description: String
		var steps: [String]
		/// Per-point, NOT shared: p6 is an independent identity pair from p1-p5 (which all
		/// share one pair — p4/p5 fork from p2's/p3's own persisted state), so these cannot be
		/// top-level singular fields the way a single shared pair's ids could be.
		var initiatorId: String
		var invitationId: String
		var dedicatedId: String
		var initiator: SideInfo
		var acceptor: SideInfo
	}
	var generatorTag: String
	var generatorCommit: String
	var bindingContractVersion: Int
	var xcframeworkProvenance: String
	var xcframeworkZipSha256Local: String
	var xcframeworkZipSha256ReleasedAsset: String
	var harnessPath: String
	/// Whether the p3 A.4 leg (EK -> CT) was driven to completion (now REQUIRED, not
	/// conditional — see `#require` below).
	var a4LegDriven: Bool
	var points: [String: PointInfo]
}

// MARK: - Shared round helpers

/// One committing round: `peer` offers an Upd, `binder` approves it and its next
/// `prepareToEncrypt` reports `didCommit`.
func committingRound(binder: PQSession, peer: PQSession) throws {
	_ = try peer.prepareToEncrypt(proposing: nil)
	let upd = try peer.encrypt(appMessage: Data("upd".utf8))
	let decrypted = try #require(try binder.decrypt(upd.cipherText))
	let offered = try #require(decrypted.proposal)
	try binder.queueProposal(digest: offered.digest)
	let prepared = try #require(try binder.prepareToEncrypt(proposing: nil))
	#expect(prepared.didCommit)
	let commitFrame = try binder.encrypt(appMessage: Data("commit".utf8))
	_ = try peer.decrypt(commitFrame.cipherText)
}

/// >= 2 messages each way, including one committing round each way — the "ordinary traffic
/// keeps flowing" half of the p4/p5 stalled-round scenario.
func driveSteadyTraffic(alice: PQSession, bob: PQSession) throws {
	try alice.exchange(with: bob)
	try committingRound(binder: alice, peer: bob)
	try committingRound(binder: bob, peer: alice)
}

private func sideInfo(
	_ session: PQSession, coreSeq: UInt64?, checkpointSeq: UInt64,
	roundInFlight: String?, legParked: Bool
) -> FixtureMeta.SideInfo {
	.init(
		coreSeq: coreSeq, checkpointSeq: checkpointSeq,
		isFullyEstablished: session.isFullyEstablished,
		turn: session.turn == .weInitiate ? "weInitiate" : "theyInitiate",
		roundInFlight: roundInFlight, legParked: legParked)
}

			description: "forked from p2's own persisted rows: alice began the A.3 bootstrap "

				+ "complete A.3, so it never reaches p3/p5/p6. >= 2 messages each way (including "

			steps: [

				"alice: finishBootstrap(rotating: nil) — KP' never delivered to bob",
				"alice.exchange(with: bob) — plain message each way",
				"committingRound(binder: alice, peer: bob)",
				"committingRound(binder: bob, peer: alice)",
			],
			initiatorId: initiatorId, invitationId: invitationId, dedicatedId: dedicatedId,
			initiator: sideInfo(
				alice, coreSeq: aliceSnap.coreSeq, checkpointSeq: aliceSnap.checkpointSeq,

	/// p5-a4-stalled: fork fresh sessions from p3's own persisted rows (already fully

	private func makeStalledA4Point(
		aliceRow: PQSession.Persisted, bobRow: PQSession.Persisted, outDir: URL,
		initiatorId: String, invitationId: String, dedicatedId: String
	) throws -> FixtureMeta.PointInfo {

		try driveSteadyTraffic(alice: alice, bob: bob)

		let aliceSnap = try aliceSink.snapshot()
		let bobSnap = try bobSink.snapshot()
		let otherLegParked = other.pendingSideBand(sealing: .fresh) != nil

		try writePointRows(
			"p5-a4-stalled", aliceRow: aliceSnap.persisted, bobRow: bobSnap.persisted, outDir: outDir)

		let turnHolderInfo = sideInfo(
			turnHolder,
			coreSeq: bobHoldsTurn ? bobSnap.coreSeq : aliceSnap.coreSeq,
			checkpointSeq: bobHoldsTurn ? bobSnap.checkpointSeq : aliceSnap.checkpointSeq,
			roundInFlight: "A.4", legParked: legParkedOnTurnHolder)
		let otherInfo = sideInfo(
			other,
			coreSeq: bobHoldsTurn ? aliceSnap.coreSeq : bobSnap.coreSeq,
			checkpointSeq: bobHoldsTurn ? aliceSnap.checkpointSeq : bobSnap.checkpointSeq,
			roundInFlight: nil, legParked: otherLegParked)

		return FixtureMeta.PointInfo(
			description: "forked from p3's own persisted rows (already fully established — side-band "
				+ "DID carry A.3's Welcome' to get here, since full establishment needs it). Side-band "
				+ "carried A.3, then stopped: the turn holder's next ordinary send auto-staged a new "
				+ "A.4 EK, and it is parked and never delivered. >= 2 messages each way (including "
				+ "one committing round each way) ran while it sat parked.",
			steps: [
				"fork: PQSession(persisted:) from p3's initiator/acceptor rows, fresh sinks",
				"turn holder: send(to: other) — auto-stages an A.4 EK, never delivered",
				"alice.exchange(with: bob) — plain message each way",
				"committingRound(binder: alice, peer: bob)",
				"committingRound(binder: bob, peer: alice)",
			],
			initiatorId: initiatorId, invitationId: invitationId, dedicatedId: dedicatedId,
			initiator: bobHoldsTurn ? otherInfo : turnHolderInfo,
			acceptor: bobHoldsTurn ? turnHolderInfo : otherInfo)
	}

	/// p6-a3-responded-stalled: the shape a REAL deployed card session is actually in, not p4's
	/// contrived one. At `reply` time the initiator also calls `bootstrapEnvelope()` and sends
	/// that `0x13` KP' envelope as an ORDINARY queued message (piggybacked, not the retained
	/// `finishBootstrap` slot p4 uses) — the acceptor's `decodeHeader` resolves it to `.forward`
	/// (the invitation pinned `H(KP') -> spawned group` at `receive`, so `receive` must run
	/// FIRST) and the spawned session's `forwarded(...)` answers A.3 on the acceptor's side,
	/// parking the acceptor's Welcome' response. The app never transmits that response. A fresh,
	/// standalone pair — not a fork of another point.
	private func makeResponderStalledA3Point(outDir: URL) throws -> FixtureMeta.PointInfo {
		let local = try ClientWrapper()  // alice / initiator
		let remote = try ClientWrapper()  // bob / acceptor

		let (localSessionRaw, welcome, myKeyPackage, bootstrapKpCommitment) =
			try local.client.reply(keyPackageMessage: remote.currentInvitation.encodedKeyPackage)
		let aliceSink = RecordingSink()
		try localSessionRaw.installSink(aliceSink)

		let appWelcomePayload = TestAppWelcomeCodec.encode(
			sendGroupWelcome: welcome,
			remoteKeyPackage: myKeyPackage,
			bootstrapKpCommitment: bootstrapKpCommitment,
			remoteClientId: try local.clientId)
		let (localSession, encryptedCombinedWelcome) = try local.client.createTwoMLSGroup(
			remoteAgentId: try remote.clientId,
			mySendGroup: localSessionRaw,
			theirKeyPackageMessage: remote.currentInvitation.encodedKeyPackage,
			appWelcome: appWelcomePayload)

		// Pre-establishment: alice ships her pre-committed A.3 KP' as an ordinary piggybacked
		// send (the app path — NOT `finishBootstrap`, which p4 exercises instead).
		let bootstrapEnvelopeBytes = try #require(try localSession.bootstrapEnvelope())

		let dedicatedId: ClientID = .mock()
		let decoded = try remote.currentInvitation.decodeHeader(ciphertext: encryptedCombinedWelcome)
		guard case .appWelcome(let welcomeToken, let appWelcomeBytes, let stapledPrivateMessage) = decoded
		else {
			Issue.record("expected an .appWelcome decode result")
			throw TestErrors.unexpected
		}
		let unpacked = try TestAppWelcomeCodec.decode(appWelcomeBytes)
		let (remoteSession, _) = try remote.currentInvitation.receive(
			sendGroupWelcome: unpacked.sendGroupWelcome,
			remoteKeyPackage: unpacked.remoteKeyPackage,
			bootstrapKpCommitment: unpacked.bootstrapKpCommitment,
			remoteClientId: unpacked.remoteClientId,
			welcomeToken: welcomeToken,
			stapledMessage: stapledPrivateMessage,
			newClientId: dedicatedId)
		let bobSink = RecordingSink()
		try remoteSession.installSink(bobSink)

		// Deliver the piggybacked 0x13 envelope AFTER receive (receive is what pins the
		// forward-table entry `decodeHeader` resolves this against — it cannot come first).
		let bootstrapDecoded = try remote.currentInvitation.decodeHeader(
			ciphertext: bootstrapEnvelopeBytes)
		guard case .forward(_, let mlsMessageData) = bootstrapDecoded else {
			Issue.record("expected a .forward decode result for the piggybacked bootstrap KP'")
			throw TestErrors.unexpected
		}
		// Answers A.3 on bob's side (parks his Welcome' response) — never delivered.
		_ = try remoteSession.forwarded(headerDecrypted: mlsMessageData)

		try remoteSession.installMockEstablishmentEnvelope()
		try localSession.acceptEstablishment(from: remoteSession, dedicatedId: dedicatedId)

		try driveSteadyTraffic(alice: localSession, bob: remoteSession)

		let aliceSnap = try aliceSink.snapshot()
		let bobSnap = try bobSink.snapshot()
		let aliceLegParked = localSession.pendingSideBand(sealing: .fresh) != nil
		let bobLegParked = remoteSession.pendingSideBand(sealing: .fresh) != nil

		try writePointRows(
			"p6-a3-responded-stalled", aliceRow: aliceSnap.persisted, bobRow: bobSnap.persisted,
			outDir: outDir)

		return FixtureMeta.PointInfo(
			description: "the field shape: a fresh, standalone pair where alice piggybacked her "
				+ "pre-committed A.3 KP' as an ordinary queued 0x13 message (bootstrapEnvelope(), "
				+ "the real app path) rather than the explicit finishBootstrap() retained slot p4 "
				+ "exercises. Bob's decodeHeader resolved it to .forward and his session's "
				+ "forwarded(...) answered A.3, parking HIS Welcome' response — the app never "
				+ "transmits it. Alice paused/resumed on bob's first frame and folded his catch-up "
				+ "Upd; >= 2 messages each way (including a committing round each way) ran while "
				+ "bob's response sat parked.",
			steps: [
				"alice: client.reply", "alice: client.createTwoMLSGroup",
				"alice: bootstrapEnvelope() — the pre-committed A.3 KP', piggybacked",
				"bob: invitation.decodeHeader + invitation.receive(newClientId: dedicated)",
				"bob: invitation.decodeHeader(bootstrapEnvelopeBytes) -> .forward",
				"bob: session.forwarded(headerDecrypted:) — answers A.3, parks Welcome', never sent",
				"bob: installMockEstablishmentEnvelope", "bob: send first frame",
				"alice: pause on 0x0B handoff, verify, resume",
				"alice.exchange(with: bob) — plain message each way",
				"committingRound(binder: alice, peer: bob)",
				"committingRound(binder: bob, peer: alice)",
			],
			initiatorId: try local.clientId.base64EncodedString(),
			invitationId: try remote.clientId.base64EncodedString(),
			dedicatedId: dedicatedId.base64EncodedString(),
			initiator: sideInfo(
				localSession, coreSeq: aliceSnap.coreSeq, checkpointSeq: aliceSnap.checkpointSeq,
				roundInFlight: "A.3", legParked: aliceLegParked),
			acceptor: sideInfo(
				remoteSession, coreSeq: bobSnap.coreSeq, checkpointSeq: bobSnap.checkpointSeq,
				roundInFlight: "A.3", legParked: bobLegParked))
	}

	/// Shared row-writing for the forked points (p4/p5), matching the top-level `writePoint`'s
	/// file layout without needing its enclosing FixtureMeta-building closure.
	private func writePointRows(
		_ name: String, aliceRow: PQSession.Persisted, bobRow: PQSession.Persisted, outDir: URL
	) throws {
		let pointDir = outDir.appendingPathComponent(name, isDirectory: true)
		try FileManager.default.createDirectory(at: pointDir, withIntermediateDirectories: true)
		let encoder = JSONEncoder()
		try encoder.encode(aliceRow).write(to: pointDir.appendingPathComponent("initiator.json"))
		try encoder.encode(bobRow).write(to: pointDir.appendingPathComponent("acceptor.json"))
	}
